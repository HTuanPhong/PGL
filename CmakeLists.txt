cmake_minimum_required(VERSION 3.30.0)
project(pgl_apps C)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

file(GLOB APP_DIRS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/apps/*)

foreach(app ${APP_DIRS})
    get_filename_component(name ${app} NAME)
    add_executable(${name} WIN32 "${app}/${name}_main.c")
    target_include_directories(${name} PRIVATE 
        ${CMAKE_SOURCE_DIR}/pgl
        ${CMAKE_SOURCE_DIR}/third_party
    )

    # if(WIN32)
    #     target_link_libraries(${name} PRIVATE
    #         dwmapi
    #     )
    # endif()

    add_custom_target(
        "run_${name}"
        COMMAND ${CMAKE_COMMAND} -E echo "" # downline
        COMMAND "$<TARGET_FILE:${name}>"
        DEPENDS ${name}
        COMMENT "Running ${name} in $<TARGET_FILE_DIR:${name}>"
        USES_TERMINAL
        VERBATIM
    )
endforeach()
